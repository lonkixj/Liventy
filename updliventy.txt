local fastButtonsRegistry = {}

local function createFastToggleButton(featureTitle, getActiveState, toggleCallback, initialPos)
    local shortName = shortNamesMap[featureTitle] or string.sub(featureTitle, 1, 4)
    
    local fastBtn = Instance.new("TextButton")
    fastBtn.Name = "FastToggle_" .. featureTitle
    fastBtn.Size = UDim2.new(0, 38, 0, 38)
    fastBtn.Position = initialPos or UDim2.new(0, 70 + (math.random(1, 4) * 42), 0, 80)
    fastBtn.BackgroundColor3 = getActiveState() and Color3.fromRGB(35, 140, 65) or Color3.fromRGB(40, 44, 58)
    fastBtn.BackgroundTransparency = 0.45
    fastBtn.Text = shortName
    fastBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    fastBtn.TextSize = 10
    fastBtn.Font = Enum.Font.GothamBold
    fastBtn.AutoButtonColor = false
    fastBtn.Parent = screenGui

    Instance.new("UICorner", fastBtn).CornerRadius = UDim.new(1, 0)
    local fastStroke = Instance.new("UIStroke", fastBtn)
    fastStroke.Thickness = 1.2
    fastStroke.Color = Color3.fromRGB(255, 255, 255)
    fastStroke.Transparency = 0.5

    local isDragging = false
    local dragStart, startPos
    local dragThreshold = 6

    fastBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isDragging = false
            dragStart = input.Position
            startPos = fastBtn.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    if not isDragging then
                        toggleCallback()
                    end
                    isDragging = false
                    dragStart = nil
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragStart and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > dragThreshold then
                isDragging = true
            end
            if isDragging then
                fastBtn.Position = UDim2.new(
                    startPos.X.Scale,
                    startPos.X.Offset + delta.X,
                    startPos.Y.Scale,
                    startPos.Y.Offset + delta.Y
                )
            end
        end
    end)

    return fastBtn
end

local function getClosestAimTarget()
    local screenCenter = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y / 2)
    local radius = tonumber(fovInput.Text) or aimFovRadius
    local closestPart = nil
    local shortestDist = radius

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer and not aimWhitelistedPlayers[player] then
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local targetPart = char and char:FindFirstChild(aimTargetPart)

            if char and hum and hum.Health > 0 and targetPart then
                local screenPos, onScreen = camera:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local distToCenter = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                    if distToCenter <= shortestDist then
                        local isVisible = true
                        if not aimWallCheck then
                            local origin = camera.CFrame.Position
                            local direction = (targetPart.Position - origin)
                            local rayParams = RaycastParams.new()
                            rayParams.FilterType = Enum.RaycastFilterType.Exclude
                            rayParams.FilterDescendantsInstances = {localPlayer.Character, char, camera}
                            local result = workspace:Raycast(origin, direction, rayParams)
                            if result and result.Instance and result.Instance.CanCollide and result.Instance.Transparency < 0.9 then
                                isVisible = false
                            end
                        end

                        if isVisible then
                            shortestDist = distToCenter
                            closestPart = targetPart
                        end
                    end
                end
            end
        end
    end
    return closestPart
end

local moveVector = Vector2.zero
local joystickDragging = false
local joystickTouchInput = nil

local function updateJoystick(inputPosition)
    local center = joystickFrame.AbsolutePosition + (joystickFrame.AbsoluteSize / 2)
    local delta = inputPosition - center
    local radius = joystickFrame.AbsoluteSize.X / 2
    local distance = delta.Magnitude

    if distance > radius then delta = delta.Unit * radius end
    joystickKnob.Position = UDim2.new(0.5, delta.X, 0.5, delta.Y)
    moveVector = delta / radius
end

joystickFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        joystickDragging = true
        joystickTouchInput = input
        updateJoystick(Vector2.new(input.Position.X, input.Position.Y))
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if joystickDragging and (input == joystickTouchInput or input.UserInputType == Enum.UserInputType.MouseMovement) then
        updateJoystick(Vector2.new(input.Position.X, input.Position.Y))
    end
end)

local function resetJoystick()
    joystickDragging = false
    joystickTouchInput = nil
    moveVector = Vector2.zero
    joystickKnob.Position = UDim2.new(0.5, 0, 0.5, 0)
end

UserInputService.InputEnded:Connect(function(input)
    if input == joystickTouchInput or input.UserInputType == Enum.UserInputType.MouseButton1 then
        resetJoystick()
    end
end)

RunService.RenderStepped:Connect(function(deltaTime)
    if isAimEnabled then
        local currentRadius = tonumber(fovInput.Text) or aimFovRadius
        fovCircleFrame.Size = UDim2.new(0, currentRadius * 2, 0, currentRadius * 2)
        fovCircleFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
        fovCircleFrame.Visible = true

        local aimTarget = getClosestAimTarget()
        if aimTarget then
            local speed = tonumber(aimSpeedInput.Text) or aimSpeedValue
            local lerpFactor = math.clamp(speed * deltaTime * 4, 0.05, 1.0)

            local targetCFrame = CFrame.lookAt(camera.CFrame.Position, aimTarget.Position)
            camera.CFrame = camera.CFrame:Lerp(targetCFrame, lerpFactor)
        end
    else
        fovCircleFrame.Visible = false
    end

    local localChar = localPlayer.Character
    local localHrp = localChar and localChar:FindFirstChild("HumanoidRootPart")
    local maxDist = tonumber(espDistInput.Text) or 300

    for _, targetPlayer in ipairs(Players:GetPlayers()) do
        if targetPlayer ~= localPlayer then
            local data = espContainerMap[targetPlayer]
            local targetChar = targetPlayer.Character
            local targetHrp = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
            local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")

            if isEspEnabled and targetChar and targetHrp and targetHum and targetHum.Health > 0 then
                local distance = localHrp and math.floor((localHrp.Position - targetHrp.Position).Magnitude) or 0

                if distance <= maxDist then
                    if not data or not data.Highlight or data.Highlight.Parent ~= targetChar then
                        setupPlayerEsp(targetPlayer)
                        data = espContainerMap[targetPlayer]
                    end

                    if data then
                        data.Highlight.Enabled = true
                        data.Highlight.OutlineColor = currentEspColor

                        data.Billboard.Enabled = true
                        data.Billboard.MaxDistance = maxDist

                        local cardFrame = data.Billboard:FindFirstChild("CardFrame")
                        if cardFrame then
                            cardFrame.CardStroke.Color = currentEspColor
                            cardFrame.NickLabel.Text = targetPlayer.DisplayName
                            cardFrame.UserLabel.Text = "@" .. targetPlayer.Name
                            cardFrame.HpLabel.Text = "HP-" .. math.floor(targetHum.Health)
                            cardFrame.DistLabel.Text = distance .. " M"
                        end

                        targetHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
                    end
                else
                    if data then
                        data.Highlight.Enabled = false
                        data.Billboard.Enabled = false
                    end
                end
            else
                if data then
                    data.Highlight.Enabled = false
                    data.Billboard.Enabled = false
                end
            end
        end
    end
end)

-- Физика движения
RunService.Heartbeat:Connect(function(deltaTime)
    local character = localPlayer.Character
    if not character then return end

    local rootPart = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not rootPart or not humanoid then return end

    -- Combat Follow
    if isCombatFollowActive and selectedCombatPlayer then
        local targetChar = selectedCombatPlayer.Character
        local targetHrp = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")

        if targetChar and targetHrp and targetHum and targetHum.Health > 0 then
            local targetCFrame = targetHrp.CFrame * CFrame.new(0, 2.5, 2)
            rootPart.AssemblyLinearVelocity = targetHrp.AssemblyLinearVelocity
            rootPart.CFrame = CFrame.lookAt(targetCFrame.Position, targetHrp.Position)

            local tool = character:FindFirstChildOfClass("Tool") or localPlayer.Backpack:FindFirstChildOfClass("Tool")
            if tool then
                if tool.Parent ~= character then
                    humanoid:EquipTool(tool)
                end
                tool:Activate()
            end
        else
            isCombatFollowActive = false
        end
    end

    -- SpeedHack
    if isSpeedEnabled and not isFlyEnabled and not isCombatFollowActive then
        local targetSpeed = tonumber(speedInput.Text) or 50
        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0 then
            local extraSpeed = math.max(0, targetSpeed - 16)
            rootPart.CFrame = rootPart.CFrame + (moveDir * extraSpeed * deltaTime)
        end
    end

    -- Fly
    if isFlyEnabled and not isCombatFollowActive then
        local flySpeed = tonumber(flySpeedInput.Text) or 50
        rootPart.AssemblyLinearVelocity = Vector3.zero

        local camCFrame = camera.CFrame
        local forward = camCFrame.LookVector
        local right = camCFrame.RightVector

        local dir = Vector3.zero
        if moveVector.Magnitude > 0.05 then
            dir = (right * moveVector.X) + (forward * -moveVector.Y)
        end

        if dir.Magnitude > 0 then
            rootPart.CFrame = rootPart.CFrame + (dir.Unit * flySpeed * deltaTime)
        end
    end
end)

local dragging = false
local dragStartPos = Vector2.zero
local frameStartPos = UDim2.new()
local targetPos = mainFrame.Position

headerFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStartPos = Vector2.new(input.Position.X, input.Position.Y)
        frameStartPos = mainFrame.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStartPos
        targetPos = UDim2.new(
            frameStartPos.X.Scale,
            frameStartPos.X.Offset + delta.X,
            frameStartPos.Y.Scale,
            frameStartPos.Y.Offset + delta.Y
        )
    end
end)

RunService.RenderStepped:Connect(function(deltaTime)
    if dragging or (mainFrame.Position ~= targetPos) then
        mainFrame.Position = mainFrame.Position:Lerp(targetPos, math.clamp(deltaTime * 14, 0, 1))
    end
end)

local windowTweenInfo = TweenInfo.new(0.35, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
local isWindowVisible = true

local function toggleMainWindow()
    isWindowVisible = not isWindowVisible
    if isWindowVisible then
        mainFrame.Visible = true
        TweenService:Create(mainFrame, windowTweenInfo, {Size = UDim2.new(0, 340, 0, 210)}):Play()
    else
        local closeTween = TweenService:Create(mainFrame, windowTweenInfo, {Size = UDim2.new(0, 340, 0, 0)})
        closeTween:Play()
        closeTween.Completed:Connect(function()
            if not isWindowVisible then
                mainFrame.Visible = false
            end
        end)
    end
end

openToggleButton.MouseButton1Click:Connect(toggleMainWindow)
